  (children run in order; a Selector stops at the first that succeeds, a Sequence at the first that fails;
   [if ...] lines are the decorators on the node below them)
  [root if behemoth_in_state_btdeco  (State=Combat, StateRequirement=ExcludedByState)]
  [root if current_target_is_valid_btdeco  (ValidityType=Accessible)]

    1) bttask_set_boolean  (Key=bb.TempBool)
    [if ForceSuccess]
    [if Blackboard  (HasEnteredCombatSinceLastDisengage is Is Not Set)]

      1) bttask_set_boolean  (Key=bb.TempBool, Value=true)
    3) RunBehavior  (BehaviorAsset=BehaviorTree shared_enter_combat_internal_bt)
    [if ForceSuccess]
    4) generic_action_bttask  (ActionType=PreEnterCombat)

    6) Sequence
      [if Blackboard  (TempBool is Is Set)]
      1) generic_action_bttask  (ActionType=EnterCombat)
