  (children run in order; a Selector stops at the first that succeeds, a Sequence at the first that fails;
   [if ...] lines are the decorators on the node below them)
  [root if behemoth_in_state_btdeco  (State=EnteringArena)]
  Sequence
    {service hide_behemoth_bttask  (DisableCollisionDuringHide=false)}
    [if ForceSuccess]
    1) Sequence
      1) Sequence
        {service hide_behemoth_bttask  (DisableCollisionDuringHide=false)}
        1) Wait  (WaitTime=0.3)

      2) Sequence
        {service flight_navigation_btservice  (PitchRotationRateOverride=15, SpeedSmoothingOverride=0, DirectionSmoothingOverride=0, MaxFlySpeedOverride=2500, MaxFlyAccelerationOverride=5500)}
        1) FlyDirectlyTo  (AcceptableRadius=1000, bStopOnOverlapNeedsUpdate=false, BlackboardKey=bb.CurrentPatrolPoint)
      3) BehemothAction "exit fly"  (Montage=AnimMontage b_mf_loco_flee_exit_01_mot)
    2) try_set_behemoth_state_bttask  (State=EnteringArena, StateChangeType=End, WhenInterruptedChangeState=EnteringArena, InterruptedChangeStateEvent=End)
