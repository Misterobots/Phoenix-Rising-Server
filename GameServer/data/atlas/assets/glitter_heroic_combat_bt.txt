              1) perform_action_montage_bttask "Tailswipe R -> L"  (Montage=AnimMontage b_glitter_normal_atk_tailswipe_mot)

        2) Sequence "Rage Pulse"
          1) perform_action_montage_bttask "Manta -> Jelly"  (Montage=AnimMontage b_glitter_normal_state_manta_to_jelly_mot)

          4) perform_action_montage_bttask "Jelly Burst"  (Montage=AnimMontage b_glitter_normal_atk_jelly_burst_rage_mot)
          5) perform_action_montage_bttask "Jelly -> Manta"  (Montage=AnimMontage b_glitter_normal_state_jelly_to_manta_mot)

        [if Cooldown  (CooldownTime=30)]
        1) Selector
          [if Cooldown  (CooldownTime=600)]
          [if comment_btdeco "Let's not immediately execute aether stuff on aether enter"]
          1) Wait  (WaitTime=0.2)

            1) perform_action_montage_bttask "Manta -> Jelly"  (Montage=AnimMontage b_glitter_normal_state_manta_to_jelly_mot)
            2) RunBehavior  (BehaviorAsset=BehaviorTree glit_teleport_bt)
            [if ForceSuccess]

              1) perform_action_montage_bttask "Jelly Pulse"  (Montage=AnimMontage b_glitter_normal_atk_jelly_pulse_unhide_mot)

            1) perform_action_montage_bttask "Manta -> Jelly"  (Montage=AnimMontage b_glitter_normal_state_manta_to_jelly_mot)

        1) EvaluateSpecificAction  (ActionClassesToTest=["BlueprintGeneratedClass glitter_normal_action_sideflop_l_C", "BlueprintGeneratedClass glitter_normal_action_sideflop_r_C"], ChosenActionKey=bb.TempAction)

          [if ForceSuccess]
          1) RunBehavior  (BehaviorAsset=BehaviorTree glitter_normal_sideflop_attacks_bt)
          2) modify_attack_cooldown_bttask "Put Side Flop L On Cooldown"  (ActionClass=BlueprintGeneratedClass glitter_normal_action_sideflop_l_C, bShowPropertyDetails=false)
          3) modify_attack_cooldown_bttask "Put Side Flop R On Cooldown"  (ActionClass=BlueprintGeneratedClass glitter_normal_action_sideflop_r_C, bShowPropertyDetails=false)

      3) RunBehavior  (BehaviorAsset=BehaviorTree glitter_normal_knot_attacks_bt)

        [if BTDeco_ArchonCooldown  (CooldownTime=5, StartTime=30, TimerGameTimeStart=bb.CurrentFightStateStartTime)]

        3) Sequence
          1) pivot_towards_target_bttask  (PivotTarget=bb.AttackTarget, BlendOutTime=0.25)
          2) perform_action_montage_bttask  (Montage=AnimMontage b_glitter_normal_atk_manta_lazer_longrange_mot)
