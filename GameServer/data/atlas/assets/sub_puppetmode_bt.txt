sub_puppetmode_bt  blackboard=BlackboardData bb_base
  (children run in order; a Selector stops at the first that succeeds, a Sequence at the first that fails;
   [if ...] lines are the decorators on the node below them)
  [root if Blackboard  (PuppetMode is Set)]
  Selector
    [if DistanceCheck  (Origin=bb.SelfActor, Target=bb.TargetActor, MinDistance=401, MaxDistance=1000000000)]
    1) Sequence
      1) MoveToAndFace  (FacingAngleBlackboardKey=bb.TargetActor, bAlignWithFacingObject=true, AcceptableRadius=400, bReachTestIncludesGoalRadius=false, bStopOnOverlapNeedsUpdate=false, BlackboardKey=bb.TargetActor)
        {service LookAlongPath  (HelperKey=bb.TargetActor)}
      2) Wait  (WaitTime=0.5)
    2) Wait  (WaitTime=0.5)
